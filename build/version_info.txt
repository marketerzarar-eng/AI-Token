# Windows version-info resource, embedded into the .exe by PyInstaller.
# Filling this in properly is what makes Explorer's "Details" tab and the
# UAC / SmartScreen prompt show a real product name and company instead of
# a blank "Unknown" — replace CompanyName with your real legal/business
# name before shipping (this alone does not remove the SmartScreen
# warning; see PACKAGING.md for what does).

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          "040904B0",
          [
            StringStruct("CompanyName", "Your Company Name"),
            StringStruct("FileDescription", "AI Token Auditor"),
            StringStruct("FileVersion", "1.0.0.0"),
            StringStruct("InternalName", "AITokenAuditor"),
            StringStruct("LegalCopyright", "© 2026 Your Company Name. All rights reserved."),
            StringStruct("OriginalFilename", "AI Token Auditor.exe"),
            StringStruct("ProductName", "AI Token Auditor"),
            StringStruct("ProductVersion", "1.0.0.0"),
          ],
        )
      ]
    ),
    VarFileInfo([VarStruct("Translation", [1033, 1200])]),
  ],
)
